"""Fixed synthetic SQLite management benchmark; not model/token/task-quality evidence."""
import hashlib
import json
import platform
import os
from pathlib import Path
import sqlite3
import statistics
import sys
import tempfile
import time
from agent_runtime.project_memory import MemoryLedger


def run(output):
    records = []
    for layout in ('shared', 'overlap', 'unique-common'):
        with tempfile.TemporaryDirectory() as tmp:
            root = Path(tmp).resolve()
            ledger = MemoryLedger(root, create=True)
            rows = []
            expected = []
            for i in range(1600):
                scopes = (['common'] if layout == 'shared' else
                          ['common', f'bucket{i % 32}'] if layout == 'overlap' else
                          ['common', f'unique{i}'])
                timestamp = f'{i // 8:05}'
                rows.append((f'e{i:04}', 'observation', 'fixed evidence', 'fixture',
                             json.dumps(scopes), 'verified', 'active', timestamp))
                newer = 1600 - (i // 8 + 1) * 8
                if newer:
                    expected.append({'id': f'e{i:04}', 'scope': scopes,
                                     'newer_sibling_count': newer,
                                     'signal': 'newer_sibling_in_scope'})
            with ledger._connect(write=True) as db:
                db.executemany('INSERT INTO entries(id,kind,content,source,scope,evidence,status,created_at) '
                               'VALUES(?,?,?,?,?,?,?,?)', rows)
            with ledger._connect() as db:
                schema_rows = [list(r) for r in db.execute("SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type,name")]
            schema_sha = hashlib.sha256(json.dumps(schema_rows, separators=(',', ':')).encode()).hexdigest()
            fixture_sha = hashlib.sha256(json.dumps(rows, separators=(',', ':')).encode()).hexdigest()
            before = hashlib.sha256(ledger.path.read_bytes()).hexdigest()
            samples = []
            outputs = []
            for _ in range(5):
                started = time.perf_counter()
                result = ledger.reconcile(max_chars=1000000)
                elapsed = time.perf_counter() - started
                assert result['potentially_obsolete'] == expected
                assert result['active_total'] == 1600 and result['flagged'] == []
                assert result['contradictions'] == []
                samples.append(elapsed)
                # Root path varies by temporary fixture; bind complete ordered payload after normalization.
                normalized = dict(result, project_root='<fixture>')
                outputs.append(hashlib.sha256(json.dumps(normalized, sort_keys=True,
                                                        separators=(',', ':')).encode()).hexdigest())
            assert len(set(outputs)) == 1
            assert hashlib.sha256(ledger.path.read_bytes()).hexdigest() == before
            records.append({'layout': layout, 'entries': 1600, 'samples_seconds': samples,
                            'median_seconds': statistics.median(samples), 'fixture_sha256': fixture_sha,
                            'normalized_output_sha256': outputs[0], 'schema_rows': schema_rows, 'schema_sha256': schema_sha, 'db_sha256_unchanged': True,
                            'unique_scope_signatures': {'shared': 1, 'overlap': 32, 'unique-common': 1600}[layout]})
    value = {'python': sys.version, 'platform': platform.platform(), 'processor': platform.processor(), 'cpu_count': os.cpu_count(),
             'affinity': sorted(os.sched_getaffinity(0)) if hasattr(os, 'sched_getaffinity') else None, 'sqlite': sqlite3.sqlite_version,
             'scope': 'synthetic full SQLite reconcile API including hydration and JSON budget; setup/hash/oracle excluded; no models/tokens/billing/task-quality',
             'code_sha256': hashlib.sha256(Path('agent_runtime/project_memory.py').read_bytes()).hexdigest(),
             'benchmarks': records}
    Path(output).write_text(json.dumps(value, indent=2) + '\n')
    print(json.dumps(value, separators=(',', ':')))
