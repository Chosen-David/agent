"""Real managed Engine/backend/wrapper/SQLite path; synthetic host and clock."""
import json
from pathlib import Path
import sys
import unittest
from unittest.mock import patch

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))
from agent_runtime import task_manifest
from agent_runtime.core import Store
from agent_runtime.task_supervisor import ManagedEngine, load_backend
from tests import test_plan_review as fixtures

ADAPTER = '''from pathlib import Path
from agent_runtime.core import Outcome

def build(root, store, run_id):
    class Child:
        idempotent = True
        required_capabilities = frozenset()
        grant = True
        policy_calls = 0
        entries = None
        identities = None
        def run(self, task, context):
            self.entries.append({'current': context.current(), 'permitted': self.grant})
            (Path(root) / 'child-effect.txt').write_bytes(b'actual child entry')
            return Outcome('pending', 'synthetic child observation')
        def verify(self, task, evidence):
            return False
    child = Child()
    child.entries, child.identities = [], []
    def authorize(plan, task, actual_handler):
        child.policy_calls += 1
        child.identities.append(actual_handler is child)
        return child.grant and actual_handler is child
    return {'handlers': {'verify_artifacts': child}, 'authorize': authorize}
'''


class ManagedChildAdmission(unittest.TestCase):
    def boundary(self, mode):
        f = fixtures.PlanReviewTests()
        f.setUp()
        self.addCleanup(f.doCleanups)
        root = f.root
        f.approve()
        state_dir = root / 'managed-state'
        store = Store(state_dir / 'state.sqlite')
        store.create(f.plan, project_root=root)
        adapter = root / 'trusted_test_adapter.py'
        adapter.write_bytes(ADAPTER.encode('utf8'))
        config = {'run_id': 'r1', 'state_dir': str(state_dir), 'project_root': str(root),
                  'session': 'synthetic-managed-config', 'adapter': str(adapter), 'allow_legacy': False}
        handlers, authorize, _, result_verifier, _ = load_backend(config, store)
        child = handlers['verify_artifacts'].handler
        if mode == 'denied':
            child.grant = False
        events = []
        original = task_manifest._prior_result_search

        def lookup(*args, **kwargs):
            result = original(*args, **kwargs)
            events.append('lookup')
            if mode == 'revoked':
                child.grant = False
            elif mode == 'cancelled':
                store.cancel('r1', 'explicit synthetic cancellation', now=f.now[0])
            return result

        engine = ManagedEngine(config, store, handlers, authorize,
                               result_verifier=result_verifier, plan_review_verifier=f.session.verify,
                               clock=lambda: f.now[0], lease_seconds=5)
        with patch.object(task_manifest, '_prior_result_search', side_effect=lookup):
            state = engine.tick('r1', 'managed-child-boundary')
        marker = root / 'child-effect.txt'
        progress = json.loads((state_dir / 'progress.json').read_bytes())
        self.assertEqual(progress['run_id'], 'r1')
        self.assertFalse(progress['all_reportable'])
        print('MANAGED_CHILD_PROBE ' + json.dumps({'mode': mode, 'events': events,
            'entries': child.entries, 'marker_hex': marker.read_bytes().hex() if marker.exists() else None,
            'policy_calls': child.policy_calls, 'identities': child.identities,
            'status': state['status'], 'attempts': state['tasks']['T1']['attempts'],
            'published_progress_exists': (state_dir / 'progress.json').exists(),
            'published_all_reportable': progress['all_reportable']}, sort_keys=True))
        self.assertTrue(child.identities)
        self.assertTrue(all(child.identities), 'host authorization must see the actual registered child')
        if mode == 'healthy':
            self.assertEqual(events, ['lookup'])
            self.assertEqual(child.entries, [{'current': True, 'permitted': True}])
            self.assertEqual(marker.read_bytes(), b'actual child entry')
        else:
            self.assertEqual(child.entries, [], 'actual child must not enter after authority/claim invalidation')
            self.assertFalse(marker.exists())
            if mode == 'denied':
                self.assertEqual(events, [])
                self.assertEqual(state['tasks']['T1']['attempts'], 0)
            else:
                self.assertEqual(events, ['lookup'])
            if mode == 'cancelled':
                self.assertEqual(state['status'], 'cancelled')
        self.assertNotEqual(state['status'], 'done', 'pending child is never reportable completion')

    def test_healthy_child(self):
        self.boundary('healthy')

    def test_initial_denial(self):
        self.boundary('denied')

    def test_lookup_cancellation(self):
        self.boundary('cancelled')

    def test_lookup_authority_revoked(self):
        self.boundary('revoked')


if __name__ == '__main__':
    suite = unittest.defaultTestLoader.loadTestsFromTestCase(ManagedChildAdmission)
    result = unittest.TextTestRunner(verbosity=2).run(suite)
    d = Path(__file__).resolve().parent
    (d / 'diagnostic-child.json').write_text(json.dumps({'methods': result.testsRun,
        'failures': len(result.failures), 'errors': len(result.errors), 'skipped': len(result.skipped)},indent=2)+'\n')
    raise SystemExit(0 if result.wasSuccessful() else 1)
