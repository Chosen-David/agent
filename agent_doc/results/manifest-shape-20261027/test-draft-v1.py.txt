"""Malformed installation metadata at actual offline CLI entry points."""
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile
import unittest

import setup


class SetupManifestTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix='manifest shape ')
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name).resolve()
        self.project = self.root / 'project'
        self.project.mkdir()
        self.inventory = self.root / 'inventory.json'
        self.inventory.write_text(json.dumps(setup.EMPTY_INVENTORY), encoding='utf-8')
        self.manifest = self.project / '.claude/agent-workflows/installation.json'
        self.env = dict(os.environ)
        self.env.pop('ANTHROPIC_BASE_URL', None)
        self.env.pop('TMUX', None)
        self.cli('--model-inventory', str(self.inventory), expected=0)
        self.original = json.loads(self.manifest.read_bytes())

    def snapshot(self):
        result = {}
        for path in self.project.rglob('*'):
            name = path.relative_to(self.project).as_posix()
            if path.is_symlink():
                result[name] = ('link', os.readlink(path))
            elif path.is_file():
                result[name] = ('file', path.read_bytes())
            else:
                result[name] = ('dir',)
        return result

    def cli(self, *args, expected):
        result = subprocess.run(
            [sys.executable, '-B', str(setup.REPO / 'setup.py'), '--target',
             str(self.project), *args], cwd=self.root, env=self.env,
            capture_output=True, encoding='utf-8', errors='strict', timeout=20)
        self.assertEqual(result.returncode, expected, result.stdout + result.stderr)
        self.assertNotIn('Traceback', result.stderr)
        if expected == 2 and '--check' not in args and '--doctor' not in args:
            self.assertEqual(result.stdout, '')
            self.assertIn('setup blocked:', result.stderr)
        if '--check' in args or '--doctor' in args:
            self.assertEqual(json.loads(result.stdout)['status'],
                             'configured' if expected == 0 else 'needs_setup')
        return result

    def refused(self, value, commands):
        self.manifest.write_text(json.dumps(value), encoding='utf-8')
        before = self.snapshot()
        for command in commands:
            with self.subTest(command=command):
                self.cli(*command, expected=2)
                self.assertEqual(self.snapshot(), before)

    def test_top_level_nonobjects_are_controlled_and_preserved(self):
        for value in (None, [], 17, 'not an object'):
            with self.subTest(value=value):
                self.refused(value, [(), ('--check',), ('--doctor',), ('--uninstall',)])

    def test_mapping_containers_are_rejected_before_writes(self):
        for field in ('files', 'links', 'skill_copies'):
            for value in (None, [], 17, 'not a mapping'):
                with self.subTest(field=field, value=value):
                    manifest = {**self.original, field: value}
                    self.refused(manifest, [(), ('--check',), ('--doctor',), ('--uninstall',)])

    def test_present_invalid_inventory_is_not_silently_replaced(self):
        for value in (None, [], {'models': []}):
            with self.subTest(value=value):
                self.refused({**self.original, 'model_inventory': value},
                             [(), ('--model-inventory', str(self.inventory)),
                              ('--check',), ('--doctor',)])

    def test_legacy_optional_fields_and_lifecycle(self):
        manifest = dict(self.original)
        for field in ('model_inventory', 'managed_block_sha256'):
            manifest.pop(field, None)
        self.manifest.write_text(json.dumps(manifest), encoding='utf-8')
        sentinel = self.project / 'notes.txt'
        sentinel.write_bytes(b'user notes\r\n')
        self.cli(expected=0)
        self.cli('--check', expected=0)
        self.cli('--doctor', expected=0)
        self.cli('--uninstall', expected=0)
        self.assertEqual(sentinel.read_bytes(), b'user notes\r\n')
        self.assertFalse(self.manifest.exists())


if __name__ == '__main__':
    unittest.main()
