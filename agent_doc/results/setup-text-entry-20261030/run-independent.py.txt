import datetime, hashlib, json, os, platform, signal, subprocess, sys, time
from pathlib import Path
root = Path(__file__).resolve().parents[2]
run = Path(__file__).resolve().parent
deadline = datetime.datetime.fromisoformat('2026-10-30T11:40:02.112332+00:00')
now = lambda: datetime.datetime.now(datetime.timezone.utc)
if (deadline-now()).total_seconds() <= 10:
    raise SystemExit('deadline reserve unavailable')
marker = os.open(run/'independent.once', os.O_WRONLY|os.O_CREAT|os.O_EXCL, 0o600)
os.close(marker)
command = [sys.executable, '-B', '-m', 'unittest', 'tests.test_setup_text', 'tests.test_setup', 'tests.test_setup_doctor', 'tests.test_setup_manifest', 'tests.test_install_recovery', '-v']
env = dict(os.environ)
removed = sorted(set(['ANTHROPIC_BASE_URL','TMUX'] + [name for name in env if 'GATEWAY' in name.upper() or name in ('OPENAI_BASE_URL','OPENAI_API_BASE','ANTHROPIC_API_URL')]))
for name in removed:
    env.pop(name, None)
started=now(); tick=time.monotonic(); child=None; timed_out=False; killed=False; code=None
with (run/'independent.stdout').open('xb') as stdout, (run/'independent.stderr').open('xb') as stderr:
    os.chmod(stdout.name,0o600); os.chmod(stderr.name,0o600)
    try:
        child=subprocess.Popen(command,cwd=root,env=env,stdout=stdout,stderr=stderr,start_new_session=True)
        timeout=min(120,(deadline-now()).total_seconds()-10)
        try:
            code=child.wait(timeout=max(0,timeout))
        except subprocess.TimeoutExpired:
            timed_out=True; killed=True
            os.killpg(child.pid,signal.SIGKILL)
            code=child.wait(timeout=10)
    finally:
        if child is not None and child.poll() is None:
            killed=True; os.killpg(child.pid,signal.SIGKILL); code=child.wait(timeout=10)
meta={'kind':'independent','attempt':1,'max_attempts':1,'started_at':started.isoformat(),'ended_at':now().isoformat(),'command':command,'cwd':str(root),'exit_code':code,'timed_out':timed_out,'effective_timeout_seconds':timeout,'deadline':deadline.isoformat(),'cleanup_reserve_seconds':10,'owned_process_group':child.pid if child else None,'kill_sent':killed,'child_reaped':child is not None and child.returncode is not None,'environment_removed_names':removed,'python':sys.version,'platform':platform.platform(),'elapsed_seconds':time.monotonic()-tick,'timing_scope':'regression accounting only, not performance evidence'}
with (run/'independent.json').open('x') as f: json.dump(meta,f,indent=2); f.write('\n')
print(json.dumps(meta))
raise SystemExit(code)
