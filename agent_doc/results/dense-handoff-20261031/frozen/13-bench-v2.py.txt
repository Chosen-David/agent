def load(name,path):

def claims(n,dense):
 return [dict(id=f'c{i}',status='supported',knowledge_ids=[],memory_ids=[],artifact_ids=['proof'],depends_on=([f'c{j}' for j in range(i)] if dense else ([f'c{i-1}'] if i else [])),assumptions=[]) for i in range(n)]

class Fixture:
 def __init__(self,root,n,dense):
  self.root=root;root.mkdir();raw=b'{"verified_fixture":true}\n';(root/'proof.json').write_bytes(raw)
  ref=dict(id='proof',path='proof.json',sha256=hashlib.sha256(raw).hexdigest())
  self.ledger=MemoryLedger(root,create=True);self.ledger.add('m1','observation','fixture','synthetic:benchmark',evidence='verified')
  record=dict(schema_version=1,run_id='bench',role='producer',input_version='v1',status='completed',limitations=[],artifacts=[ref],checks=[dict(criterion='fixture',status='pass',artifact_ids=['proof'])],tasks=[dict(task_id='T1',status='done',evidence=['proof'])],memory_refs=['m1'],evidence_claims=claims(n,dense))
  data=json.dumps(record).encode();(root/'handoff.json').write_bytes(data)
  self.event=dict(event_id='E0',run_id='bench',input_version='v1',sender='producer',task_id='T1',kind='artifact',summary='bounded fixture',action='consume',refs=[dict(id='handoff',path='handoff.json',sha256=hashlib.sha256(data).hexdigest())])
  self.request=dict(schema_version=1,input_version='v1',tasks=[dict(task_id='T1')],memory_refs=['m1'],memory_required=True)
  self.box=Mailbox(root/'mail.sqlite',dict(schema_version=1,run_id='bench',input_version='v1',max_events=1000,routes=[dict(sender='producer',recipient='consumer',task_id='T1',kind='artifact')]),root)
  self.counter=0;self.expected=None;self.record=record
 def flow(self):
  self.counter+=1;event=dict(self.event,event_id='E'+str(self.counter));t=time.perf_counter_ns();seq=self.box.publish(event)['seq'];result=self.box.prepare_context('consumer',seq,self.request,max_chars=200000);self.box.acknowledge('consumer',seq,dict(status='consumed',reason='fixture read; not semantic acceptance'));elapsed=time.perf_counter_ns()-t
  payload=result['payload'];decoded=json.loads(payload)
  if decoded['evidence_claims']!=self.record['evidence_claims']:raise AssertionError('claim loss')
  if self.expected is None:self.expected=payload
  if payload!=self.expected:raise AssertionError('payload changed')
  return elapsed
 def stale(self):
  old_seq=self.counter;self.counter+=1
  pending=self.box.publish(dict(self.event,event_id='E'+str(self.counter)))['seq']
  self.ledger.correct('m1','m2','corrected fixture','synthetic:correction','test')
  for seq in [old_seq,pending]:
   try:self.box.prepare_context('consumer',seq,self.request,max_chars=200000)
   except ValueError:pass
   else:raise AssertionError('stale memory accepted')
  if [x['seq'] for x in self.box.status() if x['receipt'] is None]!=[pending]:raise AssertionError('receipt changed')
  if [x['seq'] for x in self.box.inbox('consumer')]!=[pending]:raise AssertionError('pending correction delivery lost')
