import hashlib
import json
from pathlib import Path
import sys


def main():
    d = Path(__file__).resolve().parent
    root = d.parents[1]
    sys.path.insert(0, str(root))
    from agent_runtime.result_store import ResultStore
    run = 'engine-pre-action-authority-20261031'
    package = root / 'agent_doc/results' / run
    before = package / 'record.json'
    index = package / 'evidence-index.json'
    baseline = json.loads((d / 'registry-baseline.json').read_bytes())
    assert all(baseline['checks'].values())
    assert not index.exists()
    assert hashlib.sha256(before.read_bytes()).hexdigest() == baseline['original_record_sha256']
    assert all(hashlib.sha256((root / name).read_bytes()).hexdigest() == value for name, value in baseline['source_hashes'].items())
    frozen = {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest() for p in (package / 'frozen').iterdir() if p.is_file()}
    assert frozen == baseline['frozen_hashes'] and len(frozen) == 59
    old = json.loads(before.read_bytes())
    review = next(x for x in old['artifacts'] if x['original_path'].endswith('/bindings/result-review-v4.json'))
    before.rename(index)
    refs = [{'path': str(index.relative_to(root)), 'sha256': hashlib.sha256(index.read_bytes()).hexdigest()},
            {'path': review['published_path'], 'sha256': review['sha256']}]
    shown = ResultStore(root).register_history(run,
        'Engine pre-action authority v4 finite108-method local/independent evidence; original revoked-authority2 negatives and withdrawn candidates retained; historical refs only, no inner-effect or model/cost claim',
        refs, context={'scope': 'Finite Engine entry authority and reviewed-input bindings; SYSTEM99 open',
                       'code_revision': '2bb689c446c4cfbc05aecbbe34a6b8b5d35bc05a'})
    (d / 'registry-registration.json').write_text(json.dumps(shown, ensure_ascii=False, indent=2) + '\n')
    print(json.dumps({'record_ref': shown['record_ref'], 'origin': shown['record']['origin'],
                      'validation': shown['record']['validation_at_registration']}))


if __name__ == '__main__':
    main()
